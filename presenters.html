function saveAsImage() { const captureElement = document.getElementById('capture-area'); const actionButtons = document.querySelector('.modal-actions'); actionButtons.style.display = 'none'; // เพิ่ม scale เป็น 4 เพื่อความคมชัดระดับสูง (HD / Retina) html2canvas(captureElement, { scale: 4, useCORS: true, backgroundColor: '#fbf9f5', logging: false }).then(canvas => { actionButtons.style.display = 'flex'; const imageURL = canvas.toDataURL('image/png', 1.0); const isIOS = /iPad|iPhone|iPod/.test(navigator.userAgent) && !window.MSStream; if (isIOS) { let existingPreview = document.getElementById('ios-preview-overlay'); if (existingPreview) existingPreview.remove(); let overlay = document.createElement('div'); overlay.id = 'ios-preview-overlay'; overlay.style.cssText = 'position:fixed; top:0; left:0; width:100vw; height:100vh; background:rgba(0,0,0,0.95); z-index:99999; display:flex; flex-direction:column; align-items:center; justify-content:center; padding:20px; box-sizing:border-box;'; overlay.innerHTML = `
กดค้างที่รูปด้านล่างเพื่อบันทึกภาพความชัดสูง
`; document.body.appendChild(overlay); } else { const link = document.createElement('a'); link.download = 'Lingling-Kwong-Brand-Endorser-2026.png'; link.href = imageURL; link.click(); } }).catch(err => { console.error('Error capturing image:', err); actionButtons.style.display = 'flex'; alert('ไม่สามารถสร้างรูปภาพได้ กรุณาลองใหม่อีกครั้ง'); }); }
